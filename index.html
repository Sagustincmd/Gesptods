SQL — tabla ventas
create table ventas ( id uuid default gen_random_uuid() primary key, fecha date not null, monto numeric not null default 0, observaciones text, usuario text, hora_carga timestamptz default now(), modificado boolean default false, created_at timestamptz default now() ); create table ventas_historial ( id uuid default gen_random_uuid() primary key, venta_id uuid references ventas(id), valor_anterior numeric, valor_nuevo numeric, motivo text, usuario text, fecha_modificacion timestamptz default now() ); alter table ventas enable row level security; create policy "public access" on ventas for all using (true); alter table ventas_historial enable row level security; create policy "public access" on ventas_historial for all using (true);
